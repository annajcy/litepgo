add_library(litepgo_common INTERFACE)
add_library(litepgo::common ALIAS litepgo_common)
target_include_directories(litepgo_common INTERFACE "${PROJECT_BINARY_DIR}/include")
if(WIN32)
    target_compile_definitions(litepgo_common INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_compile_features(litepgo_common INTERFACE cxx_std_20)
target_compile_definitions(litepgo_common INTERFACE EIGEN_DONT_PARALLELIZE LITEPGO_VERSION="${PROJECT_VERSION}")
option(LITEPGO_EIGEN_DONT_VECTORIZE "Disable Eigen SIMD in this build" OFF)
if(LITEPGO_EIGEN_DONT_VECTORIZE)
    target_compile_definitions(litepgo_common INTERFACE EIGEN_DONT_VECTORIZE)
endif()
target_link_libraries(litepgo_common INTERFACE Eigen3::Eigen TBB::tbb)
# Keep inline build metadata identical without enabling solvers on base-only consumers.
if(LITEPGO_WITH_ACCELERATE)
    target_compile_definitions(litepgo_common INTERFACE LITEPGO_CONFIGURED_ACCELERATE)
endif()
if(LITEPGO_WITH_MKL)
    target_compile_definitions(litepgo_common INTERFACE LITEPGO_CONFIGURED_MKL)
endif()
if(LITEPGO_WITH_IPC_TOOLKIT)
    target_compile_definitions(litepgo_common INTERFACE LITEPGO_IPC_TOOLKIT_VERSION="${LITEPGO_IPC_TOOLKIT_VERSION}")
endif()
